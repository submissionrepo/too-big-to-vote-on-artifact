% \lane{colour}{y0}{y1}{chain}{token}: a swimlane with a dark tab
\newcommand{\lane}[5]{%
  \fill[#1!7, rounded corners=4pt] (0,#2) rectangle (\lanew,#3);
  \begin{scope}
    \clip[rounded corners=4pt] (0,#2) rectangle (\lanew,#3);
    \fill[#1] (0,#2) rectangle (\tabw,#3);
  \end{scope}
  \draw[#1!60, line width=0.4pt, rounded corners=4pt] (0,#2) rectangle (\lanew,#3);
  \pic at ({0.5*\tabw}, {#3-0.33}) {blocksicon};
  \node[rotate=90, text=white, font=\scriptsize, align=center, inner sep=0pt]
    at ({0.5*\tabw}, {0.5*(#2+#3-0.62)}) {\textbf{#4}\\[-1pt]#5};
}
% \cardface{node}{lane colour}{icon}{title}: card with shadow and a tinted header band
\newcommand{\cardface}[4]{%
  \fill[otteal, rounded corners=3pt] ([shift={(\shadow,-\shadow)}]#1.south west) rectangle ([shift={(\shadow,-\shadow)}]#1.north east);
  \fill[white, rounded corners=3pt] (#1.south west) rectangle (#1.north east);
  \begin{scope}
    \clip[rounded corners=3pt] (#1.south west) rectangle (#1.north east);
    \fill[#2!18] ([yshift=-\headh]#1.north west) rectangle (#1.north east);
  \end{scope}
  \draw[otgray, line width=0.35pt] ([yshift=-\headh]#1.north west) -- ([yshift=-\headh]#1.north east);
  \draw[otgray, line width=0.75pt, rounded corners=3pt] (#1.south west) rectangle (#1.north east);
  \pic at ([xshift=0.26cm, yshift=-0.5*\headh]#1.north west) {#3};
  \node[title] at ([xshift=0.50cm, yshift=-0.5*\headh]#1.north west) {#4};
}
