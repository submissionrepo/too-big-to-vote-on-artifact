\documentclass[border=4pt]{standalone}
% Adapted from OpenTikZ templates/flowchart (CC0). Palette names kept, mapped to the grayscale used in this paper.
% Sized so that its natural width is close to one IEEE column: the text stays legible after \includegraphics.
\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta, shapes.geometric, shapes.misc, calc}
\usepackage{mathptmx}  % Times, to match the IEEEtran body text

\definecolor{otblue}{gray}{0.42}
\definecolor{otorange}{gray}{0.10}
\definecolor{otteal}{gray}{0.68}
\definecolor{otpurple}{gray}{0.10}
\definecolor{otgray}{gray}{0.30}

\begin{document}
% ==== parameters =========================================================
\def\rowsep{0.45}
\def\boxminw{3.2}
\def\boxminh{0.75}
\def\startlabel{Market question expires\\(Polymarket, on Polygon)}
\def\readlabel{Proposer posts an answer with a bond\\(Optimistic Oracle, 2-hour window)}
\def\checklabel{Disputed?}
\def\proclabel{Vote of UMA stakers on Ethereum\\commit 24 h, then reveal 24 h;\\resolves iff the top answer holds\\$>65\%$ of all staked UMA, else rolls}
\def\endlabel{Market pays 0 / 1\\(or 50 / 50)}
\def\errorlabel{Answer\\accepted}
\def\yeslabel{yes}
\def\nolabel{no}
% =========================================================================

\begin{tikzpicture}[
    node distance=\rowsep cm,
    >={Stealth[length=2.0mm]},
    base/.style={
      line width=0.6pt, align=center, font=\small,
      minimum width=\boxminw cm, minimum height=\boxminh cm, inner xsep=4pt,
    },
    terminal/.style={base, shape=rounded rectangle, draw=otgray, fill=otteal!25},
    process/.style={base, rounded corners=2pt, draw=otgray, fill=otblue!18},
    decision/.style={base, shape=diamond, aspect=2.4, inner sep=0.5pt,
      draw=otorange, fill=white, minimum width=2.0cm, minimum height=0.8cm},
    flow/.style={draw=otgray, line width=0.7pt, ->},
    loop/.style={draw=otpurple, line width=0.7pt, ->, dashed},
    elabel/.style={font=\footnotesize, text=otgray},
  ]

  \node[terminal] (start) {\startlabel};
  \node[process,  below=of start] (read)  {\readlabel};
  \node[decision, below=of read]  (check) {\checklabel};
  \node[process,  below=of check] (proc)  {\proclabel};
  \node[terminal, below=of proc]  (end)   {\endlabel};
  \node[process, minimum width=1.5cm] (error) at (check -| proc.east) [xshift=0.2cm, anchor=west] {\errorlabel};

  \draw[flow] (start) -- (read);
  \draw[flow] (read)  -- (check);
  \draw[flow] (check) -- (proc) node[midway, right, elabel] {\yeslabel};
  \draw[flow] (proc)  -- (end);
  \draw[flow] (check) -- (error) node[midway, above, elabel] {\nolabel};
  \draw[flow] (error.south) |- (end.east);
  % reset loop: routed outside the proposer box, so the arrow never crosses its text
  \path (read.west) ++(-0.35,0) coordinate (lx);
  \draw[loop] (check.west) -- (check.west -| lx) -- (read.west -| lx) -- (read.west);
  \node[elabel, left, align=right, text=otpurple] at ($(check.west -| lx)!0.5!(read.west -| lx)$)
    {first\\dispute\\resets\\once};
\end{tikzpicture}
\end{document}
